function switchView(view) {
document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
document.querySelectorAll('.view-section').forEach(s => s.classList.remove('active'));
if(view === 'customer') {
document.querySelector('button[onclick*="customer"]').classList.add('active');
document.getElementById('customer-view').classList.add('active');
renderMenu();
} else {
document.querySelector('button[onclick*="dashboard"]').classList.add('active');
document.getElementById('dashboard-view').classList.add('active');
renderDashboard();
}
}
function renderMenu(filter = 'all') {
const grid = document.getElementById('menu-grid');
grid.innerHTML = '';
strainsData.forEach((strain, index) => {
if(filter !== 'all' && strain.type !== filter) return;
const card = document.createElement('div');
card.className = 'strain-card';
card.innerHTML = <div class="strain-img-container"> <img src="${strain.img}" class="strain-img" alt="${strain.name}"> <span class="badge">${strain.type}</span> </div> <div class="card-body"> <h2 class="strain-title">${strain.name}</h2> <div class="potency"> <span>🧬 THC: ${strain.thc}</span> <span>🌱 CBD: ${strain.cbd}</span> </div> <div class="price-display" id="price-${index}">$45</div> <div class="weight-selector"> <button class="weight-btn active" onclick="updatePriceDisplay(${index}, 3.5, this)">3.5g</button> <button class="weight-btn" onclick="updatePriceDisplay(${index}, 7, this)">7g</button> <button class="weight-btn" onclick="updatePriceDisplay(${index}, 14, this)">14g</button> <button class="weight-btn" onclick="updatePriceDisplay(${index}, 28, this)">28g</button> </div> <div class="compliance-box"> <p>GOVERNMENT WARNING: THIS PRODUCT CONTAINS CANNABIS. KEEP OUT OF REACH OF CHILDREN AND ANIMALS. FOR MEDICAL OR ADULT-USE ONLY.</p> </div> </div>;
grid.appendChild(card);
});
}
function updatePriceDisplay(cardIndex, weight, btn) {
const container = btn.parentElement;
container.querySelectorAll('.weight-btn').forEach(b => b.classList.remove('active'));
btn.classList.add('active');
document.getElementById(price-${cardIndex}).innerText = $${priceMatrix[weight]};
}
function filterStrains(type, btn) {
document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
btn.classList.add('active');
renderMenu(type);
}
function renderDashboard() {
const tbody = document.getElementById('dashboard-tbody');
tbody.innerHTML = '';
let combinedWeightGrams = 0;
strainsData.forEach((strain, index) => {
combinedWeightGrams += strain.stock;
const activeLowStockTrigger = strain.stock < 50; // Threshold logic requested (under 50 grams highlights soft red)
const row = document.createElement('tr');
if(activeLowStockTrigger) row.className = 'low-stock';
row.innerHTML = `
${strain.id}
${strain.name}
${strain.type}
${strain.stock}g
${activeLowStockTrigger ? '⚠️ LOW STOCK ALERT' : '✅ Active/Healthy'}
`;
tbody.appendChild(row);
});
document.getElementById('total-grams').innerText = combinedWeightGrams + "g";
document.getElementById('total-value').innerText = "$" + (combinedWeightGrams * ounceBulkPerGramCost).toFixed(2);
}
function executeStockAdjustment(index, type) {
const targetedInput = document.getElementById(amt-${index});
const adjustmentVolume = parseFloat(targetedInput.value);
if(isNaN(adjustmentVolume) || adjustmentVolume <= 0) return;
if(type === 'add') {
strainsData[index].stock += adjustmentVolume;
} else {
strainsData[index].stock = Math.max(0, strainsData[index].stock - adjustmentVolume);
}
renderDashboard();
}
